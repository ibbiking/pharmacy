# 4) Generic bulk import scheduler (every 15 minutes)
# Bulk selected generic products are queued and imported by scheduler.
# Single product import remains immediate.

# Cron/Scheduler ENV variables (current values)
# Add these to .env (or confirm they already exist):
# PASSWORD_RESET_CODE_EXPIRY=15
# PASSWORD_RESET_RESEND_COOLDOWN=60

# Manual run:
php artisan generic-products:process-bulk-imports

# Expected output:
# Processed X batch | Imported: Y | Skipped: Z

# Verify schedule:
php artisan schedule:list

# 5) CRON JOB COMMANDS (Local + Server)

# LOCAL (for testing scheduler without waiting for system cron)
# Run scheduler worker in foreground:
php artisan schedule:work

# OR run one-time scheduler tick manually:
php artisan schedule:run
php artisan generic-products:process-bulk-imports

# SERVER (required in production)
# Open crontab:
crontab -e

# Add this line (replace project path):
# * * * * * cd /path/to/your/project && php artisan schedule:run >> /dev/null 2>&1

# Optional log-enabled variant:
# * * * * * cd /path/to/your/project && php artisan schedule:run >> storage/logs/scheduler.log 2>&1
